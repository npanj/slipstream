Request 
struct Request {
  uint64_t  request_id;
  uint8_t   priority;      // 0 fg,1 normal,2 bg
  uint8_t   cohort;
  uint8_t   constraint_mode;
  uint8_t   reserved;
  uint64_t  deadline_ms;   // monotonic
  uint32_t  prompt_tokens;
  uint32_t  max_new_tokens;
  uint32_t  image_spans;
  float     temperature;
  float     top_p;
  uint32_t  top_k;
  uint64_t  seed;
  uint8_t   reserved2;
  // + prompt_tokens × u32
  // + image_spans × ImageSpan
};
Tokens 
struct Tokens {
  uint64_t  request_id;
  uint32_t  token_count;
  uint32_t  reserved;
  // + token_count × u32 ids
};
Done 
struct Done {
  uint64_t  request_id;
  uint8_t   finish_reason; // stop/len/err
  uint8_t   reserved;
  uint32_t  total_tokens;
  uint32_t  reserved2;
  uint64_t  gpu_seconds_ns;
  uint64_t  wall_seconds_ns;
};

Image Span

struct ImageSpan {
  uint32_t  offset;     // token offset in prompt
  uint32_t  tokens;     // number of image tokens
  uint32_t  gridHeight;
  uint32_t  gridWidth;
  uint64_t  digestLo;   // content digest (cache key)
  uint64_t  digestHi;
};

Transport FdTransport (C++)

  • Wraps a pair of POSIX fds (pipe)
  • Input fd is non-blocking only while run() is active
  • Each iteration: drain ALL available frames, then advance the backend (scheduler tick)
  • Constraint-mask responses can arrive while the target-forward command is in flight
  • Cancellation is safe because provisional writes commit only after that command drains
Why binary, not JSON? Decode emits a frame per token (or per small batch). A binary frame is a fixed 24-byte header plus raw u32 ids — no parsing cost on the hot path. JSON would add serialization overhead to every token of every request.
